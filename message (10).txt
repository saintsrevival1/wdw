-- // Utilities
local function Create(ClassName, Properties, Children)
	local Object = Instance.new(ClassName)

	for Property, Value in Properties do
		if Property ~= "Parent" then
			Object[Property] = Value
		end
	end

	if Children then
		for _, Child in Children do
			Child.Parent = Object
		end
	end

	if Properties.Parent then
		Object.Parent = Properties.Parent
	end

	return Object
end

local function Merge(Defaults, Overrides)
	local Result = {}

	for Key, Value in Defaults do
		Result[Key] = Value
	end

	for Key, Value in Overrides do
		Result[Key] = Value
	end

	return Result
end

local function Tween(Object, Information, Goal)
	local NewTween = TweenService:Create(Object, Information, Goal)
	NewTween:Play()
	return NewTween
end

local function Connect(Signal, Callback)
	local Connection = Signal:Connect(Callback)
	table.insert(Library.Connections, Connection)
	return Connection
end

local function SafeCall(Callback, ...)
	if typeof(Callback) ~= "function" then return end

	task.spawn(function(...)
		local Success, Message = pcall(Callback, ...)

		if not Success then
			warn("[UI] Callback error: " .. tostring(Message))
		end
	end, ...)
end

local function RoundNumber(Value, Decimals)
	local Power = 10 ^ (Decimals or 0)
	return math.round(Value * Power) / Power
end

local function FormatNumber(Value, Decimals)
	return string.format("%." .. tostring(Decimals or 0) .. "f", Value)
end

local function ToHex(Color)
	return string.format("#%02X%02X%02X", math.round(Color.R * 255), math.round(Color.G * 255), math.round(Color.B * 255))
end

local function FromHex(Text)
	local Red, Green, Blue = string.match(Text, "^%s*#?(%x%x)(%x%x)(%x%x)%s*$")

	if not Red then return nil end

	return Color3.fromRGB(tonumber(Red, 16), tonumber(Green, 16), tonumber(Blue, 16))
end

local function EscapeRichText(Text)
	Text = string.gsub(Text, "&", "&amp;")
	Text = string.gsub(Text, "<", "&lt;")
	Text = string.gsub(Text, ">", "&gt;")
	return Text
end

local function GetInputPosition(Input)
	if Input.UserInputType ~= Enum.UserInputType.Touch then
		return UserInputService:GetMouseLocation() + MouseOffset
	end

	local Inset = GuiService:GetGuiInset()
	return Vector2.new(Input.Position.X + Inset.X, Input.Position.Y + Inset.Y)
end

local function IsPointInside(Object, Point)
	local Position = Object.AbsolutePosition
	local Size = Object.AbsoluteSize

	return Point.X >= Position.X and Point.X <= Position.X + Size.X
		and Point.Y >= Position.Y and Point.Y <= Position.Y + Size.Y
end

local function IsPressInput(Input)
	return Input.UserInputType == Enum.UserInputType.MouseButton1 or Input.UserInputType == Enum.UserInputType.Touch
end

local function CalibrateMouse(Object, ScreenGui)
	local Raw = UserInputService:GetMouseLocation()
	local Candidates = {Vector2.new(0, 0), -GuiService:GetGuiInset(), ScreenGui.AbsolutePosition}
	local Fits = {}

	for _, Offset in Candidates do
		if IsPointInside(Object, Raw + Offset) then
			local Known = false

			for _, Fit in Fits do
				if (Fit - Offset).Magnitude < 0.5 then
					Known = true
				end
			end

			if not Known then
				table.insert(Fits, Offset)
			end
		end
	end

	if #Fits == 1 then
		MouseOffset = Fits[1]
	end
end

local function GetIcon(Name)
	if not IconLookup then
		IconLookup = {}

		for AssetId, Data in IconSheets do
			for Entry in string.gmatch(Data, "[^;]+") do
				local IconName, X, Y = string.match(Entry, "^(.-),(%d+),(%d+)$")

				IconLookup[IconName] = {
					Image = "rbxassetid://" .. AssetId,
					ImageRectOffset = Vector2.new(tonumber(X), tonumber(Y)),
					ImageRectSize = Vector2.new(48, 48),
					Lucide = true,
				}
			end
		end
	end

	Name = string.lower(Name)

	return IconLookup[Name] or IconLookup[IconAliases[Name] or ""]
end

local function ResolveIcon(Icon, RectOffset, RectSize)
	if typeof(Icon) == "number" then
		Icon = "rbxassetid://" .. tostring(Icon)
	end

	if typeof(Icon) ~= "string" or Icon == "" then return nil end

	if string.find(Icon, "://", 1, true) then
		return {
			Image = Icon,
			ImageRectOffset = RectOffset or Vector2.new(0, 0),
			ImageRectSize = RectSize or Vector2.new(0, 0),
			Lucide = false,
		}
	end

	local Found = GetIcon(Icon)

	if not Found then
		warn("[UI] Unknown icon \"" .. Icon .. "\"")
	end

	return Found
end

local function CreateGlow(Parent, Transparency, Blur)
	local Success, Glow = pcall(Instance.new, "UIShadow")

	if not Success or not Glow then return nil end

	pcall(function()
		Glow.Color = Theme.Accent
		Glow.Transparency = Transparency or 0.5
		Glow.BlurRadius = UDim.new(0, Blur or 8)
		Glow.Spread = UDim2.new(0, 1, 0, 1)
		Glow.Offset = UDim2.new(0, 0, 0, 0)
		Glow.Enabled = true
		Glow.Parent = Parent
	end)

	return Glow
end

local function GetKeyName(Key)
	if not Key then return "None" end
	return KeyNames[Key.Name] or Key.Name
end

local function KeyMatches(Key, Input)
	if not Key then return false end

	if Key.EnumType == Enum.KeyCode then
		return Input.KeyCode == Key
	end

	return Input.UserInputType == Key
end

local function BeginDrag(Input, Drag)
	Drag.Mouse = Input.UserInputType ~= Enum.UserInputType.Touch
	ActiveDrag = Drag
	Drag.Update(GetInputPosition(Input))
end

local function EndDrag()
	local Drag = ActiveDrag
	if not Drag then return end

	ActiveDrag = nil

	if Drag.Finish then
		Drag.Finish()
	end
end

local function ClosePopup()
	if not ActivePopup then return end

	local Popup = ActivePopup
	ActivePopup = nil
	Popup.Dismiss.Visible = false
	Popup.Close()
end

local function ReleasePopup(Frame)
	if ActivePopup and ActivePopup.Frame == Frame then
		ActivePopup.Dismiss.Visible = false
		ActivePopup = nil
	end
end

local function PositionPopup(Popup)
	local Anchor = Popup.Anchor
	local Parent = Popup.Frame.Parent
	local AnchorPosition = Anchor.AbsolutePosition
	local AnchorSize = Anchor.AbsoluteSize
	local Origin = Parent.AbsolutePosition
	local Bounds = Parent.AbsoluteSize

	local X = AnchorPosition.X + AnchorSize.X - Popup.Width
	local Y = AnchorPosition.Y + AnchorSize.Y + 4

	if Y + Popup.Height > Origin.Y + Bounds.Y - 8 then
		Y = AnchorPosition.Y - Popup.Height - 4
	end

	X = math.clamp(X, Origin.X + 8, math.max(Origin.X + 8, Origin.X + Bounds.X - Popup.Width - 8))

	Popup.Frame.Position = UDim2.fromOffset(X - Origin.X, Y - Origin.Y)
end

local function OpenPopup(Popup)
	ClosePopup()
	ActivePopup = Popup
	Popup.Dismiss.Visible = true
	PositionPopup(Popup)
end

local function InitializeInput()
	if InputInitialized then return end
	InputInitialized = true

	Connect(UserInputService.InputChanged, function(Input)
		if ActiveDrag and not ActiveDrag.Mouse and Input.UserInputType == Enum.UserInputType.Touch then
			ActiveDrag.Update(GetInputPosition(Input))
		end
	end)

	Connect(UserInputService.InputEnded, function(Input)
		if ActiveDrag and IsPressInput(Input) then
			EndDrag()
		end
	end)

	Connect(UserInputService.WindowFocusReleased, EndDrag)

	Connect(RunService.RenderStepped, function()
		if not ActiveDrag or not ActiveDrag.Mouse then return end

		if not UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) then
			EndDrag()
			return
		end

		ActiveDrag.Update(UserInputService:GetMouseLocation())
	end)

	Connect(UserInputService.InputBegan, function(Input)
		if Library.Binding or UserInputService:GetFocusedTextBox() then return end

		for _, Window in Library.Windows do
			if KeyMatches(Window.ToggleKey, Input) then
				Window:Toggle()
			end
		end

		for _, Keybind in Library.Keybinds do
			if KeyMatches(Keybind.Value, Input) then
				SafeCall(Keybind.Pressed, Keybind.Value)
			end
		end
	end)

	Connect(RunService.RenderStepped, function()
		if not ActivePopup then return end

		local Anchor = ActivePopup.Anchor
		local Center = Anchor.AbsolutePosition + Anchor.AbsoluteSize / 2

		if not IsPointInside(ActivePopup.ClipFrame, Center) then
			ClosePopup()
			return
		end

		PositionPopup(ActivePopup)
	end)
end

		Tween(Wrapper, Animations.Smooth, {Size = UDim2.new(1, 0, 0, 0)})
		task.wait(0.3)
		Wrapper:Destroy()
	end)

function Library.Unload()
	Library.Unloaded = true
	ClosePopup()
	ActiveDrag = nil

	for _, Connection in Library.Connections do
		Connection:Disconnect()
	end

	for _, Window in Library.Windows do
		Window.ScreenGui:Destroy()
	end

	if Library.NotificationGui then
		Library.NotificationGui:Destroy()
	end

	table.clear(Library.Connections)
	table.clear(Library.Windows)
	table.clear(Library.Keybinds)
	InputInitialized = false
	NotificationHolder = nil
end

		return Root.AbsolutePosition + Root.AbsoluteSize / 2 - ScreenGui.AbsolutePosition

		local Bounds = ScreenGui.AbsoluteSize
		local Half = Root.AbsoluteSize / 2 + Vector2.new(Layout.ShadowSpread, Layout.ShadowSpread)

		return Vector2.new(
			math.clamp(Center.X, Half.X, math.max(Half.X, Bounds.X - Half.X)),
			math.clamp(Center.Y, Half.Y, math.max(Half.Y, Bounds.Y - Half.Y))
		)
	end

		local GrabOffset = ToGuiSpace(GetInputPosition(Input)) - StartCenter

		Root.Position = UDim2.fromOffset(StartCenter.X, StartCenter.Y)
		Window.GoalPosition = StartCenter

		BeginDrag(Input, {
			Update = function(Position)
				Window.GoalPosition = ClampCenter(ToGuiSpace(Position) - GrabOffset)
			end,
			Finish = function()
				local Goal = Window.GoalPosition
				if Goal then
					Root.Position = UDim2.fromOffset(Goal.X, Goal.Y)
					Window.GoalPosition = nil
				end
			end,
		})
	end)

		local Goal = Window.GoalPosition
		if not Goal then return end

		local Current = Vector2.new(Root.Position.X.Offset, Root.Position.Y.Offset)
		local Next = Current:Lerp(Goal, 1 - math.exp(-DeltaTime * Layout.DragSpeed))

		if (Goal - Next).Magnitude < 0.5 then
			Next = Goal
			Window.GoalPosition = nil
		end

		Root.Position = UDim2.fromOffset(Next.X, Next.Y)
	end)

			Root.Position = UDim2.fromOffset(Clamped.X, Clamped.Y)
		end
	end)

	Window:SetVisible(true)

	return Window

function WindowClass.SetVisible(Self, State)
	if Self.Visible == State then return end

	Self.Visible = State
	ClosePopup()
	Self.Tooltip.Visible = false

	Self.OpenButton.Visible = not State and (UserInputService.TouchEnabled or not Self.ToggleKey)

	Tween(Self.Shadow, State and Animations.Page or Animations.Exit, {
		ImageTransparency = State and Layout.ShadowTransparency or 1,
	})

	if State then
		Self.Root.Visible = true
		Tween(Self.Scale, Animations.Dramatic, {Scale = 1})

		Self.Logo.Rotation = -200
		Tween(Self.Logo, Animations.Dramatic, {Rotation = 0})

		Self.Veil.BackgroundTransparency = 0
		Tween(Self.Veil, Animations.Page, {BackgroundTransparency = 1})

		if Self.ActiveTab then
			Self.ActiveTab:Reveal()
		end
	else
		Tween(Self.Logo, Animations.Close, {Rotation = 200})
		Tween(Self.Veil, Animations.Veil, {BackgroundTransparency = 0})

		Tween(Self.Scale, Animations.Close, {Scale = 0.7}).Completed:Connect(function()
			if not Self.Visible then
				Self.Root.Visible = false
			end
		end)
	end
end

function WindowClass.Toggle(Self)
	Self:SetVisible(not Self.Visible)
end

function WindowClass.SetToggleKey(Self, Key)
	Self.ToggleKey = Key
end

function WindowClass.SelectTab(Self, Tab)
	if Self.ActiveTab == Tab then return end

	ClosePopup()

	local Previous = Self.ActiveTab

	if Previous then
		Previous.Page.Visible = false
		Tween(Previous.Button, Animations.Smooth, {BackgroundTransparency = 1})
		Tween(Previous.ButtonStroke, Animations.Smooth, {Transparency = 1})

		if Previous.ButtonGlow then
			Tween(Previous.ButtonGlow, Animations.Smooth, {Transparency = 1})
		end
		Tween(Previous.Icon, Animations.Smooth, {[Previous.ColorProperty] = Theme.SubText})
		Tween(Previous.Label, Animations.Smooth, {TextColor3 = Theme.SubText})
		Tween(Previous.Bar, Animations.Smooth, {BackgroundTransparency = 1, Size = UDim2.fromOffset(3, 0)})
	end

	Self.ActiveTab = Tab

	Tween(Tab.Button, Animations.Smooth, {BackgroundTransparency = 0})
	Tween(Tab.ButtonStroke, Animations.Smooth, {Transparency = 0.55})

	if Tab.ButtonGlow then
		Tween(Tab.ButtonGlow, Animations.Smooth, {Transparency = 0.75})
	end
	Tween(Tab.Icon, Animations.Smooth, {[Tab.ColorProperty] = Theme.Accent})
	Tween(Tab.Label, Animations.Smooth, {TextColor3 = Theme.Text})
	Tween(Tab.Bar, Animations.Spring, {BackgroundTransparency = 0, Size = UDim2.fromOffset(3, 16)})

	Self.Path.Text = string.format(
		"%s  <font color=\"%s\">/</font>  <font color=\"%s\">%s</font>",
		EscapeRichText(Self.Title),
		ToHex(Theme.SubText),
		ToHex(Theme.Accent),
		EscapeRichText(Tab.Name)
	)

	Tab.Page.Position = UDim2.fromOffset(0, 30)
	Tab.Page.Visible = true
	Tween(Tab.Page, Animations.Dramatic, {Position = UDim2.fromOffset(0, 0)})
	Tab:Reveal()

	Self.Cover.BackgroundTransparency = 0
	Tween(Self.Cover, Animations.Page, {BackgroundTransparency = 1})
end

		Columns = Columns,
		SectionCount = 0,
		Sections = {},
	}, TabClass)

	table.insert(Self.Tabs, Tab)

	if Category then
		table.insert(Category.Tabs, Tab)
	end

	Button.MouseButton1Click:Connect(function()
		Self:SelectTab(Tab)
	end)

	Button.MouseEnter:Connect(function()
		if Self.ActiveTab ~= Tab then
			Tween(Icon, Animations.Fast, {[ColorProperty] = Theme.Text})
			Tween(Label, Animations.Fast, {TextColor3 = Theme.Text})
		end
	end)

	Button.MouseLeave:Connect(function()
		if Self.ActiveTab ~= Tab then
			Tween(Icon, Animations.Fast, {[ColorProperty] = Theme.SubText})
			Tween(Label, Animations.Fast, {TextColor3 = Theme.SubText})
		end
	end)

	if Index == 1 then
		Self:SelectTab(Tab)
	end

	return Tab
end

local function RevealSection(Entry, Delay)
	Entry.Frame.BackgroundTransparency = 1
	Entry.Stroke.Color = Theme.Accent

	TweenService:Create(Entry.Frame, TweenInfo.new(0.45, Enum.EasingStyle.Quint, Enum.EasingDirection.Out, 0, false, Delay), {
		BackgroundTransparency = 0,
	}):Play()

	TweenService:Create(Entry.Stroke, TweenInfo.new(0.8,Enum.EasingStyle.Quint, Enum.EasingDirection.Out, 0, false, Delay), {
		Color = Theme.Stroke,
	}):Play()
end

function TabClass.Reveal(Self)
	for Index, Entry in Self.Sections do
		RevealSection(Entry, (Index - 1) * 0.05)
	end
end

function SectionClass.AddSlider(Self, SliderOptions)
	local Flag = SliderOptions.Flag or SliderOptions.Name
	local Minimum = SliderOptions.Min or 0
	local Maximum = SliderOptions.Max or 100
	local Rounding = SliderOptions.Rounding or 0
	local Suffix = SliderOptions.Suffix or ""
	local Value = math.clamp(RoundNumber(SliderOptions.Default or Minimum, Rounding), Minimum, Maximum)
	local Page = Self.Tab.Page

	local Row = CreateRow(Self, SliderOptions.Name, 42)

	local function GetAlpha(Number)
		if Maximum == Minimum then return 0 end
		return math.clamp((Number - Minimum) / (Maximum - Minimum), 0, 1)
	end

	CreateLabel({
		Name = "SliderName",
		Size = UDim2.new(1, -80, 0, 24),
		Text = SliderOptions.Name,
		Parent = Row,
	})

	local ValueBox = Create("TextBox", {
		Name = "SliderValue",
		AnchorPoint = Vector2.new(1, 0),
		BackgroundTransparency = 1,
		Position = UDim2.fromScale(1, 0),
		Size = UDim2.fromOffset(76, 24),
		ClearTextOnFocus = false,
		FontFace = Fonts.Regular,
		Text = FormatNumber(Value, Rounding) .. Suffix,
		TextColor3 = Theme.SubText,
		TextSize = 14,
		TextXAlignment = Enum.TextXAlignment.Right,
		ZIndex = 6,
		Parent = Row,
	})

	local Track = Create("Frame", {
		Name = "SliderLine",
		BackgroundColor3 = Theme.Element,
		BorderSizePixel = 0,
		Position = UDim2.fromOffset(0, 31),
		Size = UDim2.new(1, 0, 0, 4),
		Parent = Row,
	}, {
		Corner(1),
	})

	local Fill = Create("Frame", {
		Name = "SliderBar",
		BackgroundColor3 = Color3.fromRGB(255, 255, 255),
		BorderSizePixel = 0,
		Size = UDim2.fromScale(GetAlpha(Value), 1),
		Parent = Track,
	}, {
		Corner(1),
		Create("UIGradient", {
			Color = ColorSequence.new(Theme.AccentDark, Theme.Accent),
		}),
	})

	CreateGlow(Fill, 0.45, 8)

	local Knob = Create("Frame", {
		Name = "Box",
		AnchorPoint = Vector2.new(0.5, 0.5),
		BackgroundColor3 = Theme.Text,
		Position = UDim2.fromScale(GetAlpha(Value), 0.5),
		Size = UDim2.fromOffset(14, 8),
		ZIndex = 2,
		Parent = Track,
	}, {
		Corner(4),
	})

	local Hitbox = CreateHitbox({
		Position = UDim2.fromOffset(0, 24),
		Size = UDim2.new(1, 0, 0, 18),
		Parent = Row,
	})

	local Object = {Type = "Slider"}

	Library.Flags[Flag] = Value

	function Object.Set(_, NewValue, SkipCallback, Instant)
		NewValue = math.clamp(RoundNumber(NewValue, Rounding), Minimum, Maximum)

		local Changed = NewValue ~= Value
		local Alpha = GetAlpha(NewValue)

		Value = NewValue
		Library.Flags[Flag] = Value
		ValueBox.Text = FormatNumber(Value, Rounding) .. Suffix

		if Instant then
			Knob.Position = UDim2.fromScale(Alpha, 0.5)
			Fill.Size = UDim2.fromScale(Alpha, 1)
		else
			Tween(Knob, Animations.Fast, {Position = UDim2.fromScale(Alpha, 0.5)})
			Tween(Fill, Animations.Fast, {Size = UDim2.fromScale(Alpha, 1)})
		end

		if Changed and not SkipCallback then
			SafeCall(SliderOptions.Callback, Value)
		end
	end

	function Object.Get()
		return Value
	end

	Hitbox.InputBegan:Connect(function(Input)
		if not IsPressInput(Input) or ActiveDrag then return end

		ClosePopup()
		Page.ScrollingEnabled = false
		Tween(Knob, Animations.Fast, {Size = UDim2.fromOffset(24, 13), BackgroundColor3 = Theme.Accent})
		Tween(ValueBox, Animations.Fast, {TextColor3 = Theme.Text})

		local function Update(Position)
			local Alpha = math.clamp((Position.X - Track.AbsolutePosition.X) / math.max(Track.AbsoluteSize.X, 1), 0, 1)
			Object:Set(Minimum + (Maximum - Minimum) * Alpha)
		end

		BeginDrag(Input, {
			Update = Update,
			Finish = function()
				Page.ScrollingEnabled = true
				Tween(Knob, Animations.Spring, {Size = UDim2.fromOffset(14, 8), BackgroundColor3 = Theme.Text})
				Tween(ValueBox, Animations.Smooth, {TextColor3 = Theme.SubText})
			end,
		})
	end)

	ValueBox.Focused:Connect(function()
		ValueBox.Text = FormatNumber(Value, Rounding)
		Tween(ValueBox, Animations.Fast, {TextColor3 = Theme.Text})
	end)

	ValueBox.FocusLost:Connect(function()
		local Number = tonumber((string.gsub(ValueBox.Text, "[^%d%.%-]", "")))

		Tween(ValueBox, Animations.Smooth, {TextColor3 = Theme.SubText})

		if Number then
			Object:Set(Number)
		end

		ValueBox.Text = FormatNumber(Value, Rounding) .. Suffix
	end)

	return Object
end

function SectionClass.AddDropdown(Self, DropdownOptions)
	local Flag = DropdownOptions.Flag or DropdownOptions.Name
	local Multi = DropdownOptions.Multi == true
	local Options = table.clone(DropdownOptions.Options or {})
	local Window = Self.Window
	local Selected = {}
	local Items = {}
	local Open = false

	local Row = CreateRow(Self, DropdownOptions.Name)

	CreateLabel({
		Name = "DropdownName",
		Size = UDim2.new(1, -(Layout.DropdownWidth + 10), 1, 0),
		Text = DropdownOptions.Name,
		Parent = Row,
	})

	local Well = Create("Frame", {
		Name = "DropdownHolder",
		AnchorPoint = Vector2.new(1, 0.5),
		BackgroundColor3 = Theme.Element,
		Position = UDim2.fromScale(1, 0.5),
		Size = UDim2.fromOffset(Layout.DropdownWidth, Layout.WellHeight),
		Parent = Row,
	}, {
		Corner(6),
		Stroke(Theme.StrokeLight, 1),
	})

	local Display = CreateLabel({
		Name = "Display",
		Position = UDim2.fromOffset(8, 0),
		Size = UDim2.new(1, -30, 1, 0),
		Text = "None",
		TextColor3 = Theme.SubText,
		TextSize = 13,
		Parent = Well,
	})

	local Chevron = CreateIcon(GetIcon("chevron-down"), {
		Name = "Chevron",
		AnchorPoint = Vector2.new(0.5, 0.5),
		ImageColor3 = Theme.SubText,
		Position = UDim2.new(1, -13, 0.5, 0),
		Size = UDim2.fromOffset(14, 14),
		Parent = Well,
	})

	local WellHitbox = CreateHitbox({Parent = Well})
	AddHover(WellHitbox, Well, Theme.Element, Theme.ElementHover)

	local List = Create("Frame", {
		Name = DropdownOptions.Name .. "List",
		BackgroundColor3 = Theme.Element,
		Size = UDim2.fromOffset(Layout.DropdownWidth, 0),
		ClipsDescendants = true,
		Visible = false,
		Parent = Window.Overlay,
	}, {
		Corner(6),
		Stroke(Theme.StrokeLight, 1),
	})

	local Scroller = Create("ScrollingFrame", {
		Name = "Scroller",
		BackgroundTransparency = 1,
		BorderSizePixel = 0,
		Position = UDim2.fromOffset(0, 3),
		Size = UDim2.new(1, 0, 1, -6),
		CanvasSize = UDim2.fromOffset(0, 0),
		ScrollBarThickness = 2,
		ScrollBarImageColor3 = Theme.SubText,
		ScrollingDirection = Enum.ScrollingDirection.Y,
		Parent = List,
	}, {
		Create("UIListLayout", {SortOrder = Enum.SortOrder.LayoutOrder, HorizontalAlignment = Enum.HorizontalAlignment.Center}),
	})

	local Object = {Type = "Dropdown"}

	local function GetListHeight()
		return math.min(#Options, Layout.DropdownMaxItems) * Layout.DropdownItemHeight + 6
	end

	local function GetValue()
		local Result = {}

		for _, Option in Options do
			if Selected[Option] then
				table.insert(Result, Option)
			end
		end

		if Multi then
			return Result
		end

		return Result[1]
	end

	local function Refresh()
		local Value = GetValue()
		local HasValue = Multi and #Value > 0 or (not Multi and Value ~= nil)

		Library.Flags[Flag] = Value
		Display.Text = HasValue and (Multi and table.concat(Value, ", ") or tostring(Value)) or "None"
		Tween(Display, Animations.Fast, {TextColor3 = HasValue and Theme.Text or Theme.SubText})

		for Option, Item in Items do
			local IsSelected = Selected[Option] == true

			Tween(Item.Button, Animations.Fast, {TextColor3 = IsSelected and Theme.Text or Theme.SubText})
			Tween(Item.Check, Animations.Fast, {ImageTransparency = IsSelected and 0 or 1})
		end
	end

	local function SetOpen(State)
		if Open == State then return end
		Open = State

		Tween(Chevron, Animations.Smooth, {
			Rotation = State and 180 or 0,
			ImageColor3 = State and Theme.Accent or Theme.SubText,
		})

		if State then
			local Height = GetListHeight()

			List.Size = UDim2.fromOffset(Layout.DropdownWidth, 0)
			List.Visible = true
			Tween(List, Animations.Spring, {Size = UDim2.fromOffset(Layout.DropdownWidth, Height)})

			OpenPopup({
				Frame = List,
				Anchor = Well,
				Dismiss = Window.Dismiss,
				ClipFrame = Window.Holder,
				Width = Layout.DropdownWidth,
				Height = Height,
				Close = function()
					SetOpen(false)
				end,
			})
		else
			ReleasePopup(List)

			Tween(List, Animations.Fast, {Size = UDim2.fromOffset(Layout.DropdownWidth, 0)}).Completed:Connect(function()
				if not Open then
					List.Visible = false
				end
			end)
		end
	end

	local function BuildItems()
		for _, Item in Items do
			Item.Button:Destroy()
		end

		table.clear(Items)

		for Index, Option in Options do
			local Button = Create("TextButton", {
				Name = tostring(Option),
				BackgroundColor3 = Theme.AccentDark,
				BackgroundTransparency = 1,
				BorderSizePixel = 0,
				Size = UDim2.new(1, -8, 0, Layout.DropdownItemHeight),
				AutoButtonColor = false,
				FontFace = Fonts.Regular,
				Text = tostring(Option),
				TextColor3 = Theme.SubText,
				TextSize = 13,
				TextTruncate = Enum.TextTruncate.AtEnd,
				TextXAlignment = Enum.TextXAlignment.Left,
				LayoutOrder = Index,
				Parent = Scroller,
			}, {
				Corner(6),
				Create("UIPadding", {PaddingLeft = UDim.new(0, 10), PaddingRight = UDim.new(0, 26)}),
			})

			local Check = CreateIcon(GetIcon("check"), {
				Name = "Check",
				AnchorPoint = Vector2.new(1, 0.5),
				ImageColor3 = Theme.Accent,
				ImageTransparency = 1,
				Position = UDim2.new(1, 18, 0.5, 0),
				Size = UDim2.fromOffset(12, 12),
				Parent = Button,
			})

			Items[Option] = {Button = Button, Check = Check}

			Button.MouseEnter:Connect(function()
				if not Selected[Option] then
					Tween(Button, Animations.Fast, {TextColor3 = Theme.Text})
				end
			end)

			Button.MouseLeave:Connect(function()
				if not Selected[Option] then
					Tween(Button, Animations.Fast, {TextColor3 = Theme.SubText})
				end
			end)

			local LastChosen = 0

			local function Choose()
				if not Open or os.clock() - LastChosen < 0.25 then return end
				LastChosen = os.clock()

				if Multi then
					Selected[Option] = not Selected[Option] or nil
				else
					table.clear(Selected)
					Selected[Option] = true
					SetOpen(false)
				end

				Refresh()
				SafeCall(DropdownOptions.Callback, GetValue())
			end

			Button.InputBegan:Connect(function(Input)
				if Input.UserInputType == Enum.UserInputType.MouseButton1 then
					Choose()
				end
			end)

			Button.MouseButton1Click:Connect(Choose)
		end

		Scroller.CanvasSize = UDim2.fromOffset(0, #Options * Layout.DropdownItemHeight)
	end

	function Object.Set(_, NewValue, SkipCallback)
		table.clear(Selected)

		if typeof(NewValue) == "table" then
			for _, Option in NewValue do
				if table.find(Options, Option) then
					Selected[Option] = true

					if not Multi then break end
				end
			end
		elseif NewValue ~= nil and table.find(Options, NewValue) then
			Selected[NewValue] = true
		end

		Refresh()

		if not SkipCallback then
			SafeCall(DropdownOptions.Callback, GetValue())
		end
	end

	function Object.Get()
		return GetValue()
	end

	function Object.Refresh(_, NewOptions, KeepSelection)
		SetOpen(false)
		Options = table.clone(NewOptions)

		if KeepSelection then
			for Option in Selected do
				if not table.find(Options, Option) then
					Selected[Option] = nil
				end
			end
		else
			table.clear(Selected)
		end

		BuildItems()
		Refresh()
	end

	WellHitbox.MouseButton1Click:Connect(function()
		if #Options == 0 then return end
		SetOpen(not Open)
	end)

	BuildItems()
	Object:Set(DropdownOptions.Default, true)

	return Object
end

		Create("UISizeConstraint", {MinSize = Vector2.new(36, Layout.WellHeight)}),
	})

	local ButtonStroke = Stroke(Theme.StrokeLight, 1)
	ButtonStroke.Parent = Button

	AddHover(Button, Button, Theme.Element, Theme.ElementHover)

	local Object = {
		Type = "Keybind",
		Value = KeybindOptions.Default,
		Pressed = KeybindOptions.Pressed,
	}

	Library.Flags[Flag] = Object.Value
	table.insert(Library.Keybinds, Object)

	function Object.Set(_, NewKey, SkipCallback)
		Object.Value = NewKey
		Library.Flags[Flag] = NewKey
		Button.Text = GetKeyName(NewKey)

		if not SkipCallback then
			SafeCall(KeybindOptions.Callback, NewKey)
		end
	end

	function Object.Get()
		return Object.Value
	end

	Button.MouseButton1Click:Connect(function()
		if Library.Binding then return end

		Library.Binding = true
		Button.Text = "..."
		Tween(Button, Animations.Fast, {TextColor3 = Theme.Accent})
		Tween(ButtonStroke, Animations.Fast, {Color = Theme.Accent})

		local Connection

		Connection = UserInputService.InputBegan:Connect(function(Input)
			local NewKey

			if Input.KeyCode ~= Enum.KeyCode.Unknown then
				if Input.KeyCode ~= Enum.KeyCode.Escape and Input.KeyCode ~= Enum.KeyCode.Backspace then
					NewKey = Input.KeyCode
				end
			elseif Input.UserInputType == Enum.UserInputType.MouseButton2 or Input.UserInputType == Enum.UserInputType.MouseButton3 then
				NewKey = Input.UserInputType
			else
				return
			end

			Connection:Disconnect()

			Tween(Button, Animations.Smooth, {TextColor3 = Theme.SubText})
			Tween(ButtonStroke, Animations.Smooth, {Color = Theme.StrokeLight})
			Object:Set(NewKey)

			task.defer(function()
				Library.Binding = false
			end)
		end)
	end)

	return Object
end

function SectionClass.AddTextBox(Self, TextBoxOptions)
	local Flag = TextBoxOptions.Flag or TextBoxOptions.Name
	local Row = CreateRow(Self, TextBoxOptions.Name)

	CreateLabel({
		Name = "TextBoxName",
		Size = UDim2.new(1, -(Layout.DropdownWidth + 10), 1, 0),
		Text = TextBoxOptions.Name,
		Parent = Row,
	})

	local Well = Create("Frame", {
		Name = "Background",
		AnchorPoint = Vector2.new(1, 0.5),
		BackgroundColor3 = Theme.Element,
		Position = UDim2.fromScale(1, 0.5),
		Size = UDim2.fromOffset(Layout.DropdownWidth, Layout.WellHeight),
		ClipsDescendants = true,
		Parent = Row,
	}, {
		Corner(6),
	})

	local WellStroke = Stroke(Theme.StrokeLight, 1)
	WellStroke.Parent = Well

	local Input = Create("TextBox", {
		Name = "TextInput",
		BackgroundTransparency = 1,
		Position = UDim2.fromOffset(8, 0),
		Size = UDim2.new(1, -16, 1, 0),
		ClearTextOnFocus = false,
		FontFace = Fonts.Regular,
		PlaceholderColor3 = Theme.SubText,
		PlaceholderText = TextBoxOptions.Placeholder or "...",
		Text = TextBoxOptions.Default or "",
		TextColor3 = Theme.Text,
		TextSize = 13,
		TextXAlignment = Enum.TextXAlignment.Left,
		Parent = Well,
	})

	local Object = {Type = "TextBox"}

	Library.Flags[Flag] = Input.Text

	function Object.Set(_, NewText, SkipCallback)
		Input.Text = tostring(NewText)
		Library.Flags[Flag] = Input.Text

		if not SkipCallback then
			SafeCall(TextBoxOptions.Callback, Input.Text, false)
		end
	end

	function Object.Get()
		return Input.Text
	end

	Input.Focused:Connect(function()
		Tween(WellStroke, Animations.Fast, {Color = Theme.Accent})
		Tween(Well, Animations.Fast, {BackgroundColor3 = Theme.ElementHover})
	end)

	Input.FocusLost:Connect(function(EnterPressed)
		Tween(WellStroke, Animations.Smooth, {Color = Theme.StrokeLight})
		Tween(Well, Animations.Smooth, {BackgroundColor3 = Theme.Element})

		Library.Flags[Flag] = Input.Text
		SafeCall(TextBoxOptions.Callback, Input.Text, EnterPressed)
	end)

	return Object
end

		})
	end

		Name = "Button",
		AnchorPoint = Vector2.new(1, 0.5),
		BackgroundColor3 = Theme.Element,
		Position = UDim2.fromScale(1, 0.5),
		Size = Compact and UDim2.fromOffset(96, Layout.WellHeight) or UDim2.new(1, 0, 0, 26),

		TextSize = 13,
		Parent = Row,
	}, {
		Corner(6),
		Stroke(Theme.StrokeLight, 1),

		CreateIcon(ButtonIcon, {
			Name = "Icon",

		Button.BackgroundColor3 = Theme.AccentDark
		Button.TextColor3 = Theme.Accent
		Tween(Button, Animations.Page, {BackgroundColor3 = Theme.ElementHover, TextColor3 = RestColor})

		SafeCall(ButtonOptions.Callback)
	end)

	return {Type = "Button"}
end

function SectionClass.AddColorPicker(Self, ColorPickerOptions)
	local Flag = ColorPickerOptions.Flag or ColorPickerOptions.Name
	local Window = Self.Window
	local Value = ColorPickerOptions.Default or Theme.Accent
	local Hue, Saturation, Brightness = Value:ToHSV()
	local Open = false
	local PopupSize = Vector2.new(196, 176)

	local Row = CreateRow(Self, ColorPickerOptions.Name)

	CreateLabel({
		Name = "ColorPickerName",
		Size = UDim2.new(1, -44, 1, 0),
		Text = ColorPickerOptions.Name,
		Parent = Row,
	})

	local Swatch = Create("TextButton", {
		Name = "Swatch",
		AnchorPoint = Vector2.new(1, 0.5),
		BackgroundColor3 = Value,
		Position = UDim2.fromScale(1, 0.5),
		Size = UDim2.fromOffset(36, 20),
		AutoButtonColor = false,
		Text = "",
		Parent = Row,
	}, {
		Corner(6),
		Stroke(Theme.StrokeLight, 1),
	})

	local Popup = Create("Frame", {
		Name = ColorPickerOptions.Name .. "Picker",
		BackgroundColor3 = Theme.Section,
		Size = UDim2.fromOffset(PopupSize.X, 0),
		ClipsDescendants = true,
		Visible = false,
		Parent = Window.Overlay,
	}, {
		Corner(8),
		Stroke(Theme.StrokeLight, 1),
	})

	local Field = Create("TextButton", {
		Name = "Field",
		BackgroundColor3 = Color3.fromHSV(Hue, 1, 1),
		Position = UDim2.fromOffset(10, 10),
		Size = UDim2.fromOffset(176, 110),
		AutoButtonColor = false,
		Text = "",
		Parent = Popup,
	}, {
		Corner(6),
		Create("Frame", {
			Name = "White",
			BackgroundColor3 = Color3.fromRGB(255, 255, 255),
			BorderSizePixel = 0,
			Size = UDim2.fromScale(1, 1),
		}, {
			Corner(6),
			Create("UIGradient", {Transparency = NumberSequence.new(0, 1)}),
		}),
		Create("Frame", {
			Name = "Black",
			BackgroundColor3 = Color3.fromRGB(0, 0, 0),
			BorderSizePixel = 0,
			Size = UDim2.fromScale(1, 1),
			ZIndex = 2,
		}, {
			Corner(6),
			Create("UIGradient", {Rotation = 90, Transparency = NumberSequence.new(1, 0)}),
		}),
	})

	local FieldCursor = Create("Frame", {
		Name = "Cursor",
		AnchorPoint = Vector2.new(0.5, 0.5),
		BackgroundTransparency = 1,
		Position = UDim2.fromScale(Saturation, 1 - Brightness),
		Size = UDim2.fromOffset(10, 10),
		ZIndex = 3,
		Parent = Field,
	}, {
		Corner(1),
		Stroke(Color3.fromRGB(255, 255, 255), 1.5),
	})

	local HueKeypoints = {}

	for Index = 0, 6 do
		table.insert(HueKeypoints, ColorSequenceKeypoint.new(Index / 6, Color3.fromHSV(Index / 6, 1, 1)))
	end

	local HueBar = Create("TextButton", {
		Name = "Hue",
		BackgroundColor3 = Color3.fromRGB(255, 255, 255),
		Position = UDim2.fromOffset(10, 129),
		Size = UDim2.fromOffset(176, 8),
		AutoButtonColor = false,
		Text = "",
		Parent = Popup,
	}, {
		Corner(1),
		Create("UIGradient", {Color = ColorSequence.new(HueKeypoints)}),
	})

	local HueCursor = Create("Frame", {
		Name = "Cursor",
		AnchorPoint = Vector2.new(0.5, 0.5),
		BackgroundColor3 = Color3.fromRGB(255, 255, 255),
		Position = UDim2.fromScale(Hue, 0.5),
		Size = UDim2.fromOffset(4, 14),
		ZIndex = 2,
		Parent = HueBar,
	}, {
		Corner(1),
		Stroke(Theme.Section, 1.5),
	})

	local HexBox = Create("TextBox", {
		Name = "Hex",
		BackgroundColor3 = Theme.Element,
		Position = UDim2.fromOffset(10, 146),
		Size = UDim2.fromOffset(176, 20),
		ClearTextOnFocus = false,
		FontFace = Fonts.Regular,
		Text = ToHex(Value),
		TextColor3 = Theme.SubText,
		TextSize = 13,
		Parent = Popup,
	}, {
		Corner(5),
	})

	local Object = {Type = "ColorPicker"}

	Library.Flags[Flag] = Value

	local function Apply(SkipCallback)
		Value = Color3.fromHSV(Hue, Saturation, Brightness)
		Library.Flags[Flag] = Value

		Swatch.BackgroundColor3 = Value
		Field.BackgroundColor3 = Color3.fromHSV(Hue, 1, 1)
		FieldCursor.Position = UDim2.fromScale(Saturation, 1 - Brightness)
		HueCursor.Position = UDim2.fromScale(Hue, 0.5)
		HexBox.Text = ToHex(Value)

		if not SkipCallback then
			SafeCall(ColorPickerOptions.Callback, Value)
		end
	end

	local function SetOpen(State)
		if Open == State then return end
		Open = State

		if State then
			Popup.Size = UDim2.fromOffset(PopupSize.X, 0)
			Popup.Visible = true
			Tween(Popup, Animations.Spring, {Size = UDim2.fromOffset(PopupSize.X, PopupSize.Y)})

			OpenPopup({
				Frame = Popup,
				Anchor = Swatch,
				Dismiss = Window.Dismiss,
				ClipFrame = Window.Holder,
				Width = PopupSize.X,
				Height = PopupSize.Y,
				Close = function()
					SetOpen(false)
				end,
			})
		else
			ReleasePopup(Popup)

			Tween(Popup, Animations.Fast, {Size = UDim2.fromOffset(PopupSize.X, 0)}).Completed:Connect(function()
				if not Open then
					Popup.Visible = false
				end
			end)
		end
	end

	function Object.Set(_, NewColor, SkipCallback)
		Hue, Saturation, Brightness = NewColor:ToHSV()
		Apply(SkipCallback)
	end

	function Object.Get()
		return Value
	end

	Swatch.InputBegan:Connect(function(Input)
		if Input.UserInputType == Enum.UserInputType.MouseButton1 then
			CalibrateMouse(Swatch, Window.ScreenGui)
		end
	end)

	Swatch.MouseButton1Click:Connect(function()
		SetOpen(not Open)
	end)

	Field.InputBegan:Connect(function(Input)
		if not IsPressInput(Input) or ActiveDrag then return end

		if Input.UserInputType == Enum.UserInputType.MouseButton1 then
			CalibrateMouse(Field, Window.ScreenGui)
		end

		local function Update(Position)
			local Size = Field.AbsoluteSize
			local Relative = Position - Field.AbsolutePosition

			Saturation = math.clamp(Relative.X / math.max(Size.X, 1), 0, 1)
			Brightness = 1 - math.clamp(Relative.Y / math.max(Size.Y, 1), 0, 1)
			Apply()
		end

		BeginDrag(Input, {Update = Update})
	end)

	HueBar.InputBegan:Connect(function(Input)
		if not IsPressInput(Input) or ActiveDrag then return end

		if Input.UserInputType == Enum.UserInputType.MouseButton1 then
			CalibrateMouse(HueBar, Window.ScreenGui)
		end

		local function Update(Position)
			Hue = math.clamp((Position.X - HueBar.AbsolutePosition.X) / math.max(HueBar.AbsoluteSize.X, 1), 0, 1)
			Apply()
		end

		BeginDrag(Input, {Update = Update})
	end)

	HexBox.FocusLost:Connect(function()
		local NewColor = FromHex(HexBox.Text)

		if NewColor then
			Object:Set(NewColor)
		else
			HexBox.Text = ToHex(Value)
		end
	end)

	return Object
end


return Library
